<%*
/**
 * 复习回写模板（SM-2 一键回写）
 * 用法：打开一篇待复习笔记 → Ctrl+Shift+T → 选择「复习回写」→ 输入评分 q(0-5)
 * 效果：自动更新 frontmatter 的 review_date / repetition / interval / ease_factor / updated
 * 说明：本模板为纯执行块（<%* %>），不向正文插入任何内容，仅修改 frontmatter。
 * 对齐：知识飞轮系统/04-巩固/scripts/refresh_review_queue.py 识别的 5 字段。
 */

const fm = tp.file.frontmatter;
const qRaw = await tp.system.prompt("复习评分 q（0-5）", "4");
const q = Number(qRaw);
if (isNaN(q) || q < 0 || q > 5) {
  new Notice("❌ 无效评分（应为 0-5），已取消");
  return;
}

const today = moment();
const oldInterval = fm.interval || 1;
const oldRep = fm.repetition || 0;
const oldEF = fm.ease_factor || 2.5;

let newInterval, newRep, newEF = oldEF;
if (q < 3) {                 // 忘记/错误：interval 重置 1，repetition 不变，EF 下调
  newInterval = 1;
  newRep = oldRep;
  newEF = Math.max(1.3, +(oldEF - 0.2).toFixed(2));
} else if (oldRep === 0) {   // 首次通过：interval 固定 1，repetition → 1
  newInterval = 1;
  newRep = 1;
} else if (q === 3) {        // 勉强：interval 不变，repetition +1
  newInterval = oldInterval;
  newRep = oldRep + 1;
} else if (q === 4) {        // 较好（默认档）：interval × EF
  newInterval = Math.max(1, Math.round(oldInterval * oldEF));
  newRep = oldRep + 1;
} else {                     // q === 5 完美：interval × EF × 1.1
  newInterval = Math.max(1, Math.round(oldInterval * oldEF * 1.1));
  newRep = oldRep + 1;
}

const nextDate = today.add(newInterval, "days").format("YYYY-MM-DD");
const updated = moment().format("YYYY-MM-DDTHH:mm");

await app.fileManager.processFrontMatter(tp.file.path, {
  review_date: nextDate,
  repetition: newRep,
  interval: newInterval,
  ease_factor: newEF,
  updated: updated,
});

new Notice(`✅ 已记录 q=${q} → 下次复习 ${nextDate}（+${newInterval}天）`);
%>
