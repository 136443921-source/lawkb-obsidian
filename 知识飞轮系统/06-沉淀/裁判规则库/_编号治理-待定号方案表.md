import os, re, collections
root = "06-沉淀/裁判规则库"
pat = re.compile(r'^rule_id:\s*([^\s\[]+)', re.M)
occ = collections.defaultdict(list)
for dp, dn, fn in os.walk(root):
    if '.backup' in dp: continue
    for f in fn:
        if not f.endswith('.md'): continue
        head = open(os.path.join(dp, f), encoding='utf-8', errors='ignore').read(1200)
        m = pat.search(head)
        if m and re.match(r'^R-[A-Z]{2}-\d{3}$', m.group(1).strip()):
            occ[m.group(1).strip()].append(os.path.join(dp, f))
dup = {k: v for k, v in occ.items() if len(v) > 1}
